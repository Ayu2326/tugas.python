print("===PROGRAM KASIR SEDERHANA===")
harga = int(input("Masukkan Harga Barang: Rp "))
jumlah = int(input("Masukkan Jumlah Barang:"))

total = harga*jumlah

#percabangan diskon
if total >=100000:
    diskon = total*10/100
    print("Anda mendapatkan diskon 10%")
elif total >= 50000:
    diskon = total*5/100
else:
    diskon = 0
    
total_bayar = total - diskon
print("\n===STRUK BELANJA===")
print("Total Harga: Rp", total)
print("diskon: Rp", int(diskon))
print("Total Pembayaran: Rp", total_bayar)

uang_bayar = int(input("\nMasukkan Uang Bayar: Rp"))
if uang_bayar >= total_bayar:
    kembali = uang_bayar - total_bayar
    
else:
    print("uang anda tidak cukup! silahkan masukkan jumlah yang benar.")
print("\n====UANG KEMBALI===")
print("total bayar: Rp", int(total_bayar))
print("Uang Yang Diterima: Rp", uang_bayar)
print("kembalian: Rp", int(kembali))
